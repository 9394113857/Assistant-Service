if (Test-Path .\venv) {

}

}

# STEP 4 - CHECK PYTHON 3.14

py -3.14 --version

# STEP 5 - CREATE FRESH VIRTUAL ENVIRONMENT

py -3.14 -m venv venv

.\venv\Scripts\Activate.ps1
